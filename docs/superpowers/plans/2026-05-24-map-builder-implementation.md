# Lane Map Builder Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone React/Vite web app to visually create and export the `LANE_MAP` matrix.

**Architecture:** A single-page application using React. It maintains a 2D array of integers in state, renders them as a clickable CSS grid, and provides a utility function to serialize the state into a downloadable `.ts` file.

**Tech Stack:** React, Vite, TypeScript, Tailwind CSS.

---

## File Structure (Inside `tools/map-builder`)

- `package.json`: Vite project configuration.
- `src/App.tsx`: Main application containing the grid state and UI.
- `src/components/Grid.tsx`: The clickable grid component.
- `src/utils/export.ts`: Helper to convert matrix to TS string and trigger download.

### Task 2: Core Grid Logic & State

- [ ] **Step 1: Setup State and Grid Rendering**

Modify `tools/map-builder/src/App.tsx`:
```tsx
import { useState } from 'react'

const INITIAL_ROWS = 30;
const COLS = 7;

function App() {
  const [grid, setGrid] = useState<number[][]>(
    Array(INITIAL_ROWS).fill(0).map(() => Array(COLS).fill(0))
  );

  const handleCellClick = (r: number, c: number) => {
    const newGrid = [...grid];
    newGrid[r] = [...newGrid[r]];
    newGrid[r][c] = newGrid[r][c] >= 5 ? 0 : newGrid[r][c] + 1;
    setGrid(newGrid);
  };

  const addRow = () => {
    setGrid([...grid, Array(COLS).fill(0)]);
  };

  const removeRow = () => {
    if (grid.length > 1) {
      setGrid(grid.slice(0, -1));
    }
  };

  return (
    <div className="min-h-screen bg-gray-100 p-8 flex flex-col items-center">
      <h1 className="text-3xl font-bold mb-6">Lane Map Builder</h1>
      
      <div className="flex gap-4 mb-4">
        <button onClick={addRow} className="px-4 py-2 bg-blue-500 text-white rounded">Add Row</button>
        <button onClick={removeRow} className="px-4 py-2 bg-red-500 text-white rounded">Remove Row</button>
      </div>

      <div className="bg-white p-4 rounded shadow">
        {grid.map((row, rIdx) => (
          <div key={rIdx} className="flex gap-1 mb-1">
            {row.map((cell, cIdx) => (
              <button
                key={`${rIdx}-${cIdx}`}
                onClick={() => handleCellClick(rIdx, cIdx)}
                className={`w-10 h-10 border rounded flex items-center justify-center font-bold
                  ${cell === 0 ? 'bg-gray-50 border-gray-200 text-transparent' : 'bg-blue-600 border-blue-700 text-white'}`}
                style={{ opacity: cell === 0 ? 1 : 0.4 + (cell * 0.12) }}
              >
                {cell > 0 && `${cell}F`}
              </button>
            ))}
          </div>
        ))}
      </div>
    </div>
  )
}

export default App
```

```typescript
export function exportGridToTS(grid: number[][]) {
  const rows = grid.length;
  const cols = grid[0]?.length || 0;

  let tsContent = `// Auto-generated by Lane Map Builder\n\n`;
  tsContent += `export const LANE_MAP = [\n`;
  
  grid.forEach(row => {
    tsContent += `  [${row.join(', ')}],\n`;
  });
  
  tsContent += `];\n\n`;
  tsContent += `export const GRID_SIZE = { rows: ${rows}, cols: ${cols} };\n`;
  tsContent += `export const UNIT_SIZE = 1;\n`;

  const blob = new Blob([tsContent], { type: 'text/plain;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  
  const link = document.createElement('a');
  link.href = url;
  link.download = 'laneMap.ts';
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
}
```
